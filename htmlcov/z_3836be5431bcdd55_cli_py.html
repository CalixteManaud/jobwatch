59def parse_window(value: str) -> timedelta: 

60 """`26h` → 26 heures ; unités : m, h, d (ou j), w.""" 

61 match = _WINDOW_RE.match(value) 

62 if match is None: 

63 raise typer.BadParameter("format attendu : 26h, 7d, 2w ou 90m", param_hint="--since") 

64 amount, unit = int(match.group(1)), match.group(2).lower() 

65 window = timedelta(**{_WINDOW_UNITS[unit]: amount}) 

66 if not timedelta(0) < window <= MAX_WINDOW: 

67 raise typer.BadParameter("fenêtre entre 1 minute et 90 jours", param_hint="--since") 

68 return window

100def _offers_table(title: str, items: list[ScoredOffer]) -> Table: 

101 table = Table(title=escape(title), title_justify="left", expand=True) 

102 table.add_column("Score", justify="right", no_wrap=True) 

103 table.add_column("P", justify="center", no_wrap=True) 

104 table.add_column("Poste", ratio=3) 

105 table.add_column("Entreprise", ratio=2) 

106 table.add_column("Ville", ratio=2) 

107 table.add_column("Publiée", no_wrap=True) 

108 table.add_column("Raisons", ratio=3) 

109 for item in items: 

110 offer = item.offer 

111 excluded = item.result.excluded 

112 table.add_row( 

113 "exclue" if excluded else str(item.score), 

114 "-" if excluded else str(item.priority), 

115 f"[link={offer.url}]{escape(offer.intitule or offer.id)}[/link]", 

116 escape(offer.company), 

117 escape(item.city), 

118 format_date(offer.date_creation), 

119 escape(", ".join(str(reason) for reason in item.result.top_reasons(2))), 

120 style="red" if excluded else _PRIORITY_STYLES[item.priority], 

121 ) 

122 return table

155@app.command() 

156def run( 

157 since: SinceOption = "26h", 

158 dry_run: Annotated[ 

159 bool, 

160 typer.Option( 

161 "--dry-run", 

162 help="Appels API réels, digest affiché et fichiers écrits, mais ni état ni Discord.", 

163 ), 

164 ] = False, 

165 notify: Annotated[ 

166 bool, 

167 typer.Option( 

168 "--notify/--no-notify", 

169 help="Envoyer le digest sur Discord (--no-notify : envoi au prochain run).", 

170 ), 

171 ] = True, 

172 min_score: Annotated[ 

173 int | None, 

174 typer.Option("--min-score", min=0, max=100, help="Seuil d'envoi (défaut : config.yaml)."), 

175 ] = None, 

176 config: ConfigOption = None, 

177) -> None: 

178 """Collecte les nouvelles offres, les score et envoie le digest (premier run : --since 7d).""" 

179 window = parse_window(since) 

180 settings = _bootstrap() 

181 stats = RunStats(dry_run=dry_run) 

182 started = time.monotonic() 

183 report: RunReport | None = None 

184 exit_code = 0 

185 log.info("run.start", version=__version__, since=since, dry_run=dry_run, notify=notify) 

186 try: 

187 app_config = load_config(config or settings.config_path) 

188 settings.ft_credentials() # échoue tôt si les identifiants manquent 

189 notifier = DiscordNotifier(settings.webhook_url()) if notify and not dry_run else None 

190 try: 

191 with ( 

192 _ft_client(settings) as client, 

193 OfferStore.open(settings.state_path, readonly=dry_run) as store, 

194 ): 

195 try: 

196 report = run_pipeline( 

197 config=app_config, 

198 client=client, 

199 store=store, 

200 notifier=notifier, 

201 out_dir=settings.out_dir, 

202 options=RunOptions(window, since, dry_run, notify, min_score), 

203 stats=stats, 

204 now=datetime.now(UTC), 

205 ) 

206 finally: 

207 stats.http_requests = client.http_requests 

208 finally: 

209 if notifier is not None: 

210 notifier.close() 

211 except JobwatchError as exc: 

212 exit_code = exc.exit_code 

213 log.error("run.failed", error=str(exc), error_type=type(exc).__name__, exit_code=exit_code) 

214 finally: 

215 stats.duration_seconds = time.monotonic() - started 

216 log.info("run.summary", **stats.summary()) 

217 if settings.pushgateway_url: 

218 push_metrics(settings.pushgateway_url, stats) 

219 

220 if report is not None and dry_run: 

221 _print_dry_run(report) 

222 if exit_code: 

223 raise typer.Exit(exit_code)

226@app.command() 

227def search( 

228 keywords: Annotated[str, typer.Argument(help="Mots-clés, par exemple « kubernetes ».")], 

229 dept: Annotated[str | None, typer.Option("--dept", help="Code département, ex. 59.")] = None, 

230 since: SinceOption = "7d", 

231 limit: Annotated[int, typer.Option(min=1, max=500, help="Nombre de lignes affichées.")] = 30, 

232 config: ConfigOption = None, 

233) -> None: 

234 """Recherche ponctuelle, offres scorées affichées en tableau (aucun état modifié).""" 

235 window = parse_window(since) 

236 settings = _bootstrap() 

237 with _handle_errors("search"): 

238 app_config = load_config(config or settings.config_path) 

239 now = datetime.now(UTC) 

240 search_config = app_config.search 

241 zone = search_config.zone(dept) if dept else "France" 

242 query = SearchQuery(keywords, dept, now - window, now, search_config.contract_type) 

243 with _ft_client(settings) as client: 

244 client.authenticate() 

245 result = client.search(query) 

246 scorer = Scorer(app_config.scoring, contract_type=search_config.contract_type) 

247 items = sort_by_score([ScoredOffer(o, zone, scorer.score(o)) for o in result.offers]) 

248 title = ( 

249 f"{len(items)} offre(s) {search_config.contract_type} pour {keywords!r} ({zone}, {since})" 

250 ) 

251 if len(items) > limit: 251 ↛ 253line 251 didn't jump to line 253 because the condition on line 251 was always true

252 title += f" : {limit} premières" 

253 Console().print(_offers_table(title, items[:limit]))

257def sample( 

258 keywords: Annotated[str, typer.Option("--keywords", "-k", help="Mots-clés.")] = "devops", 

259 dept: Annotated[str, typer.Option("--dept", help="Code département.")] = "75",

261 since: SinceOption = "7d", 

262 output: Annotated[Path, typer.Option("--output", "-o", help="Fichier JSON produit.")] = Path( 

263 "tests/fixtures/ft_sample.json" 

264 ), 

265) -> None: 

266 """Enregistre une vraie réponse de l'API, anonymisée, comme fixture de test.""" 

267 window = parse_window(since) 

268 settings = _bootstrap() 

269 with _handle_errors("sample"): 

270 now = datetime.now(UTC) 

271 with _ft_client(settings) as client: 

272 raw = client.fetch_raw(SearchQuery(keywords, dept, now - window, now), count) 

273 

274 offers = anonymize_offers(raw) 

275 output.parent.mkdir(parents=True, exist_ok=True) 

276 output.write_text( 

277 json.dumps({"resultats": offers}, ensure_ascii=False, indent=2) + "\n", encoding="utf-8" 

278 ) 

279 

280 console = Console() 

281 console.print(f"{len(offers)} offre(s) anonymisée(s) enregistrée(s) dans {output}") 

282 table = Table(title="Champs attendus par les modèles", title_justify="left") 

283 table.add_column("Champ") 

284 table.add_column("Renseigné", justify="right") 

285 for field_name, present in field_presence(offers).items(): 

286 style = "green" if present == len(offers) else ("yellow" if present else "red") 

287 table.add_row(field_name, f"{present}/{len(offers)}", style=style) 

288 console.print(table) 

289 

290 invalid = 0 

291 for raw_offer in offers: 

292 try: 

293 Offer.from_api(raw_offer) 

294 except ValidationError as exc: 

295 invalid += 1 

296 console.print(f"[red]Offre invalide {raw_offer.get('id')!r}[/red] : {exc}") 

297 modelled = {info.alias or name for name, info in Offer.model_fields.items()} 

298 unmodelled = sorted({key for raw_offer in offers for key in raw_offer} - modelled) 

299 console.print(f"Offres valides : {len(offers) - invalid}/{len(offers)}") 

300 console.print(f"Champs non modélisés (ignorés) : {', '.join(unmodelled) or 'aucun'}")
